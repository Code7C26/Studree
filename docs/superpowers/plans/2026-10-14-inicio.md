/**
 * Fechas del Inicio (ver docs/features/home.md). Funciones puras.
 *
 * Los días se manejan como claves `YYYY-MM-DD` en la HORA LOCAL del navegador
 * (igual que la columna `date` de Postgres, que no tiene zona): así una fecha
 * no se corre de día por la diferencia con UTC.
 */

const WEEKDAYS = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado']
const WEEKDAYS_SHORT = ['dom', 'lun', 'mar', 'mié', 'jue', 'vie', 'sáb']
const MONTHS = [
  'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
  'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre',
]
const DAY_MS = 24 * 60 * 60 * 1000

/** @param {number} value */
const pad = (value) => String(value).padStart(2, '0')

/**
 * @param {Date} date
 * @returns {string} `YYYY-MM-DD` en hora local.
 */
export function toDateKey(date) {
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`
}

/**
 * @param {string} key - `YYYY-MM-DD`
 * @returns {Date} Medianoche local de ese día.
 */
export function parseDateKey(key) {
  const [year, month, day] = key.split('-').map(Number)
  return new Date(year, month - 1, day)
}

/**
 * @param {string} key
 * @param {number} days
 * @returns {string}
 */
export function addDays(key, days) {
  const date = parseDateKey(key)
  date.setDate(date.getDate() + days)
  return toDateKey(date)
}

/**
 * Días enteros entre hoy y `key` (negativo si ya pasó). `Math.round` absorbe
 * los días de 23 o 25 horas de los cambios de horario.
 * @param {string} key
 * @param {string} todayKey
 * @returns {number}
 */
export function daysUntil(key, todayKey) {
  return Math.round((parseDateKey(key).getTime() - parseDateKey(todayKey).getTime()) / DAY_MS)
}

/**
 * @param {number} days
 * @returns {string} "hoy", "mañana", "en 3 días", "ayer", "hace 3 días".
 */
export function formatCountdown(days) {
  if (days === 0) return 'hoy'
  if (days === 1) return 'mañana'
  if (days === -1) return 'ayer'
  return days > 0 ? `en ${days} días` : `hace ${-days} días`
}

/**
 * @param {string} iso
 * @param {Date} [now]
 * @returns {string} "recién", "hace 15 min", "hace 2 horas", "hace 3 días" o "el 01/08/2026".
 */
export function formatRelativeTime(iso, now = new Date()) {
  const date = new Date(iso)
  const minutes = Math.floor((now.getTime() - date.getTime()) / 60000)
  if (minutes < 1) return 'recién'
  if (minutes < 60) return `hace ${minutes} min`
  const hours = Math.floor(minutes / 60)
  if (hours < 24) return hours === 1 ? 'hace 1 hora' : `hace ${hours} horas`
  const days = Math.floor(hours / 24)
  if (days < 30) return days === 1 ? 'hace 1 día' : `hace ${days} días`
  return `el ${pad(date.getDate())}/${pad(date.getMonth() + 1)}/${date.getFullYear()}`
}

/**
 * Mismo formato que la barra de las hojas en La Carpeta (`lib/format/sheetDate.js`
 * de esa fase; se unifican después del merge).
 * @param {string} iso
 * @param {Date} [now]
 * @returns {string} "Martes 14/10 · 18:30" (con año si no es el actual).
 */
export function formatDateTime(iso, now = new Date()) {
  const date = new Date(iso)
  const year = date.getFullYear() === now.getFullYear() ? '' : `/${date.getFullYear()}`
  return `${WEEKDAYS[date.getDay()]} ${pad(date.getDate())}/${pad(date.getMonth() + 1)}${year} · ${pad(date.getHours())}:${pad(date.getMinutes())}`
}

/**
 * @param {string} key
 * @returns {string} "mar 20/10"
 */
export function formatShortDate(key) {
  const date = parseDateKey(key)
  return `${WEEKDAYS_SHORT[date.getDay()]} ${pad(date.getDate())}/${pad(date.getMonth() + 1)}`
}

/**
 * @param {string} key
 * @returns {string} "20/10"
 */
export function formatDayMonth(key) {
  const date = parseDateKey(key)
  return `${pad(date.getDate())}/${pad(date.getMonth() + 1)}`
}

/**
 * @param {number} year
 * @param {number} month - 0-based.
 * @returns {string} "Octubre 2026"
 */
export function formatMonthTitle(year, month) {
  return `${MONTHS[month]} ${year}`
}

/**
 * Grilla mensual del calendario del Inicio (ver docs/features/home.md).
 * Sin librerías: 6 semanas × 7 días, la semana empieza el LUNES.
 */

import { toDateKey } from './dates.js'

/** Encabezados de la grilla, de lunes a domingo. */
export const WEEKDAY_HEADERS = ['L', 'M', 'M', 'J', 'V', 'S', 'D']

/**
 * @typedef {{key: string, day: number, inMonth: boolean}} CalendarCell
 */

/**
 * Siempre 6 filas (aunque el mes entre en 5) para que el panel no cambie de
 * alto al pasar de mes.
 * @param {number} year
 * @param {number} month - 0-based.
 * @returns {CalendarCell[][]}
 */
export function buildMonthGrid(year, month) {
  const offset = (new Date(year, month, 1).getDay() + 6) % 7 // lunes = 0
  return Array.from({ length: 6 }, (_, week) =>
    Array.from({ length: 7 }, (_, weekday) => {
      const date = new Date(year, month, 1 - offset + week * 7 + weekday)
      return { key: toDateKey(date), day: date.getDate(), inMonth: date.getMonth() === month }
    }),
  )
}

/**
 * @param {number} year
 * @param {number} month - 0-based.
 * @param {number} delta - Meses a mover (negativo = atrás).
 * @returns {{year: number, month: number}}
 */
export function shiftMonth(year, month, delta) {
  const date = new Date(year, month + delta, 1)
  return { year: date.getFullYear(), month: date.getMonth() }
}

```js
/**
 * Preview de una hoja para "Seguí donde dejaste" (ver docs/features/home.md).
 * Las hojas no tienen título (La Carpeta): se muestra el primer renglón con
 * texto, sin la sintaxis de Markdown. Función pura.
 */

const FENCE = /^(```|~~~)/
const HORIZONTAL_RULE = /^([-*_]\s*){3,}$/

/**
 * Quita la sintaxis inline: `[[links]]` (también escapados), imágenes, links,
 * negrita/itálica/tachado y código. El `_` suelto se deja (`snake_case`).
 * @param {string} text
 * @returns {string}
 */
function stripInline(text) {
  return text
    .replace(/\\\[\\\[(.+?)\\\]\\\]/g, '$1')
    .replace(/\[\[(.+?)\]\]/g, '$1')
    .replace(/!\[([^\]]*)\]\([^)]*\)/g, '$1')
    .replace(/\[([^\]]+)\]\([^)]*\)/g, '$1')
    .replace(/(\*\*|__|~~|\*|`)/g, '')
    .replace(/\s+/g, ' ')
    .trim()
}

/**
 * @param {string|null|undefined} markdown
 * @param {number} [max=120]
 * @returns {string} Primer renglón con texto, cortado con "…", o '' si no hay.
 */
export function buildPreview(markdown, max = 120) {
  if (!markdown) return ''
  let inFence = false

  for (const raw of markdown.split(/\r?\n/)) {
    const line = raw.trim()
    if (FENCE.test(line)) {
      inFence = !inFence
      continue
    }
    if (inFence || !line || HORIZONTAL_RULE.test(line)) continue

    const text = stripInline(
      line
        .replace(/^#{1,6}(\s+|$)/, '')
        .replace(/^>\s*/, '')
        .replace(/^([-*+]|\d+[.)])\s+(\[[ xX]\]\s+)?/, ''),
    )
    if (text) return text.length > max ? `${text.slice(0, max - 1).trimEnd()}…` : text
  }
  return ''
}

 *
 * Las claves `YYYY-MM-DD` se comparan como strings: el orden alfabético es el
 * cronológico.
 */

import { addDays } from './dates.js'

/**
 * @typedef {{id: string, title: string, date: string, created_at?: string}} ImportantDate
 * @typedef {{id: string, text: string, due_date: string|null, done_at: string|null, created_at?: string}} Task
 */

/** @param {ImportantDate} a @param {ImportantDate} b */
const byDate = (a, b) => a.date.localeCompare(b.date) || (a.created_at ?? '').localeCompare(b.created_at ?? '')

/** @param {Task} task */
const isPending = (task) => !task.done_at

/**
 * @param {Task} task
 * @param {string} todayKey
 * @returns {boolean} Pendiente y con fecha límite anterior a hoy.
 */
export function isOverdue(task, todayKey) {
  return isPending(task) && Boolean(task.due_date) && task.due_date < todayKey
}

/**
 * Fechas de hoy en adelante, de la más cercana a la más lejana.
 * @param {ImportantDate[]} dates
 * @param {string} todayKey
 * @param {number} [limit=8]
 * @returns {ImportantDate[]}
 */
export function upcomingDates(dates, todayKey, limit = 8) {
  return dates.filter((item) => item.date >= todayKey).sort(byDate).slice(0, limit)
}

/**
 * @param {ImportantDate[]} dates
 * @param {string} todayKey
 * @returns {ImportantDate|null}
 */
export function pickNextDate(dates, todayKey) {
  return upcomingDates(dates, todayKey, 1)[0] ?? null
}

/**
 * "Esta semana" = los próximos 7 días (hoy a hoy+6).
 * @param {{dates: ImportantDate[], tasks: Task[], todayKey: string}} input
 * @returns {{dates: number, tasks: number, overdue: number}}
 */
export function buildWeekSummary({ dates, tasks, todayKey }) {
  const end = addDays(todayKey, 6)
  /** @param {string} key */
  const inWeek = (key) => key >= todayKey && key <= end
  return {
    dates: dates.filter((item) => inWeek(item.date)).length,
    tasks: tasks.filter((task) => isPending(task) && task.due_date && inWeek(task.due_date)).length,
    overdue: tasks.filter((task) => isOverdue(task, todayKey)).length,
  }
}

/** @param {number} count @param {string} singular @param {string} plural */
const plural = (count, singular, plural) => `${count} ${count === 1 ? singular : plural}`

/**
 * Texto del resumen, en partes para que la UI pinte en rojo las vencidas.
 * @param {{dates: number, tasks: number, overdue: number}} summary
 * @returns {{calm: boolean, parts: Array<{text: string, danger: boolean}>}}
 */
export function describeWeekSummary({ dates, tasks, overdue }) {
  const parts = [
    { text: plural(dates, 'fecha', 'fechas'), danger: false },
    { text: plural(tasks, 'tarea', 'tareas'), danger: false },
  ]
  if (overdue > 0) parts.push({ text: plural(overdue, 'vencida', 'vencidas'), danger: true })
  return { calm: dates === 0 && tasks === 0 && overdue === 0, parts }
}

/**
 * Pendientes: primero las que tienen fecha límite (la más próxima arriba),
 * después las que no, por orden de creación.
 * @param {Task[]} tasks
 * @returns {Task[]}
 */
export function sortPendingTasks(tasks) {
  return tasks.filter(isPending).sort((a, b) => {
    if (a.due_date && b.due_date && a.due_date !== b.due_date) return a.due_date.localeCompare(b.due_date)
    if (a.due_date && !b.due_date) return -1
    if (!a.due_date && b.due_date) return 1
    return (a.created_at ?? '').localeCompare(b.created_at ?? '')
  })
}

/**
 * Hechas: la última tildada arriba.
 * @param {Task[]} tasks
 * @returns {Task[]}
 */
export function sortDoneTasks(tasks) {
  return tasks.filter((task) => !isPending(task)).sort((a, b) => b.done_at.localeCompare(a.done_at))
}

/**
 * Qué marcar en cada día de la grilla: fechas importantes y tareas PENDIENTES
 * que vencen ese día.
 * @param {ImportantDate[]} dates
 * @param {Task[]} tasks
 * @returns {Map<string, {dates: boolean, tasks: boolean}>}
 */
export function buildDayMarkers(dates, tasks) {
  const markers = new Map()
  /** @param {string} key */
  const markerFor = (key) => {
    if (!markers.has(key)) markers.set(key, { dates: false, tasks: false })
    return markers.get(key)
  }
  for (const item of dates) markerFor(item.date).dates = true
  for (const task of tasks) if (isPending(task) && task.due_date) markerFor(task.due_date).tasks = true
  return markers
}

 * Una colección del Inicio: carga (cache primero) + `mutate` optimista.
 * @param {{list: Function}} service - Constante de módulo (identidad estable).
 */
function useCollection(service) {
  const [items, setItems] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  const reload = useCallback(async () => {
    const { data, error: loadError } = await service.list(undefined, (cached) => {
      setItems(cached)
      setLoading(false)
    })
    setItems(data ?? [])
    setError(loadError)
    setLoading(false)
  }, [service])

  useEffect(() => {
    reload()
  }, [reload])

  /**
   * @param {((current: Object[]) => Object[])|null} applyLocal - Cambio optimista (o null).
   * @param {() => Promise<{data: any, error: string|null}>} write
   * @returns {Promise<{data: any, error: string|null}>}
   */
  const mutate = useCallback(
    async (applyLocal, write) => {
      if (applyLocal) setItems(applyLocal)
      const result = await write()
      if (result.error) {
        await reload()
        setError(result.error)
        return result
      }
      setError(null)
      return result
    },
    [reload],
  )

  return { items, setItems, loading, error, mutate }
}

function TaskItem({ task, todayKey, onToggle, onRename, onDelete }) {

        <input
          value={draft}
          onChange={(event) => setDraft(event.target.value)}
          onBlur={finishEditing}
          onKeyDown={(event) => {

            if (event.key === 'Escape') {
              setDraft(task.text)
              setEditing(false)
            }

function NoteCard({ note, onUpdate, onRemove }) {
  const [editing, setEditing] = useState(false)
  const [draft, setDraft] = useState(note.body)

    else setDraft(note.body)
  }

    <li className="flex gap-1 rounded-lg border border-border bg-bg p-2">
      {editing ? (
        <textarea
          value={draft}
          onChange={(event) => setDraft(event.target.value)}
